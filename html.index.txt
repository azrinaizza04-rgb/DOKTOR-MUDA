// Web Audio API Synthesizer Engine (No external sound file dependencies)
const SoundEngine = {
  ctx: null,
  sfxMuted: false,
  bgmMuted: true,
  bgmInterval: null,

  init() {
    if (!this.ctx) {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume();
    }
  },

  playTone(freq, type = 'sine', duration = 0.1, delay = 0, gainVal = 0.1) {
    if (this.sfxMuted) return;
    this.init();
    if (!this.ctx) return;

    setTimeout(() => {
      try {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, this.ctx.currentTime);
        gain.gain.setValueAtTime(gainVal, this.ctx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.0001, this.ctx.currentTime + duration);

        osc.connect(gain);
        gain.connect(this.ctx.destination);

        osc.start();
        osc.stop(this.ctx.currentTime + duration);
      } catch (e) {}
    }, delay * 1000);
  },

  playCorrect() {
    // Cheerful ascending arpeggio
    const notes = [523.25, 659.25, 783.99, 1046.50, 1318.51];
    notes.forEach((f, i) => this.playTone(f, 'triangle', 0.18, i * 0.07, 0.15));
  },

  playWrong() {
    // Playful boing sound
    if (this.sfxMuted) return;
    this.init();
    if (!this.ctx) return;

    try {
      const osc = this.ctx.createOscillator();
      const gain = this.ctx.createGain();
      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(280, this.ctx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(120, this.ctx.currentTime + 0.3);

      gain.gain.setValueAtTime(0.18, this.ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.3);

      osc.connect(gain);
      gain.connect(this.ctx.destination);

      osc.start();
      osc.stop(this.ctx.currentTime + 0.3);
    } catch(e) {}
  },

  playClick() {
    this.playTone(600, 'sine', 0.05, 0, 0.08);
  },

  playCombo() {
    const notes = [659.25, 783.99, 987.77, 1318.51];
    notes.forEach((f, i) => this.playTone(f, 'sine', 0.15, i * 0.05, 0.2));
  },

  playTick() {
    this.playTone(880, 'sine', 0.03, 0, 0.05);
  },

  playVictory() {
    const tune = [
      { f: 523.25, d: 0.12 }, { f: 659.25, d: 0.12 }, { f: 783.99, d: 0.12 },
      { f: 1046.50, d: 0.25 }, { f: 880.00, d: 0.12 }, { f: 1046.50, d: 0.4 }
    ];
    let time = 0;
    tune.forEach(n => {
      this.playTone(n.f, 'triangle', n.d, time, 0.2);
      time += n.d + 0.02;
    });
  },

  toggleBGM() {
    this.bgmMuted = !this.bgmMuted;
    if (!this.bgmMuted) {
      this.startBGM();
    } else {
      this.stopBGM();
    }
    return !this.bgmMuted;
  },

  startBGM() {
    this.init();
    if (this.bgmInterval) clearInterval(this.bgmInterval);
    
    // Playful light background melody loop
    const melody = [523.25, 0, 659.25, 0, 783.99, 659.25, 523.25, 0];
    let step = 0;

    this.bgmInterval = setInterval(() => {
      if (this.bgmMuted) return;
      const freq = melody[step % melody.length];
      if (freq > 0) {
        this.playTone(freq, 'sine', 0.12, 0, 0.03);
      }
      step++;
    }, 250);
  },

  stopBGM() {
    if (this.bgmInterval) clearInterval(this.bgmInterval);
  }
};

const Confetti = {
  canvas: null,
  ctx: null,
  particles: [],

  init() {
    this.canvas = document.getElementById('confettiCanvas');
    this.ctx = this.canvas.getContext('2d');
    this.resize();
    window.addEventListener('resize', () => this.resize());
  },

  resize() {
    if (!this.canvas) return;
    this.canvas.width = window.innerWidth;
    this.canvas.height = window.innerHeight;
  },

  burst() {
    this.init();
    const colors = ['#ff5e7e', '#2ed573', '#fbc531', '#00a896', '#a55eea', '#ff9ff3'];
    for (let i = 0; i < 80; i++) {
      this.particles.push({
        x: window.innerWidth / 2,
        y: window.innerHeight / 2,
        vx: (Math.random() - 0.5) * 16,
        vy: (Math.random() - 0.8) * 16,
        size: Math.random() * 10 + 6,
        color: colors[Math.floor(Math.random() * colors.length)],
        rotation: Math.random() * 360,
        rSpeed: (Math.random() - 0.5) * 10,
        opacity: 1
      });
    }
    this.animate();
  },

  animate() {
    if (!this.particles.length) return;
    this.ctx.clearRect(0, 0, this.canvas.width, this.canvas.height);

    this.particles.forEach((p, idx) => {
      p.x += p.vx;
      p.y += p.vy;
      p.vy += 0.3; // Gravity
      p.rotation += p.rSpeed;
      p.opacity -= 0.015;

      this.ctx.save();
      this.ctx.globalAlpha = Math.max(0, p.opacity);
      this.ctx.translate(p.x, p.y);
      this.ctx.rotate((p.rotation * Math.PI) / 180);
      this.ctx.fillStyle = p.color;
      this.ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
      this.ctx.restore();

      if (p.opacity <= 0) {
        this.particles.splice(idx, 1);
      }
    });

    if (this.particles.length > 0) {
      requestAnimationFrame(() => this.animate());
    }
  }
};

function submitAnswer(userChoice) {
  if (!running || answeredCurrent) return;

  answeredCurrent = true;
  const q = questions[index];
  const isCorrect = (userChoice === q.correct);

  userAnswers[index] = {
    chosen: userChoice,
    correct: isCorrect,
    question: q
  };

  // Combo & Score Calculation
  if (isCorrect) {
    combo++;
    const bonus = combo > 1 ? combo * 20 : 0;
    score += (100 + bonus);
    
    updateMascot("happy");
    SoundEngine.playCorrect();
    Confetti.burst();

    if (combo > 1) {
      el("comboBadge").classList.remove("hidden");
      el("comboBadge").textContent = `🔥 COMBO x${combo}!`;
      SoundEngine.playCombo();
    }
  } else {
    combo = 0;
    el("comboBadge").classList.add("hidden");
    updateMascot("wrong");
    SoundEngine.playWrong();
  }

  el("score").textContent = `⭐ ${score}`;

  // Highlight choices visually
  if (userChoice === true) {
    el("btnFalse").classList.add("opacity-40", "scale-95");
  } else {
    el("btnTrue").classList.add("opacity-40", "scale-95");
  }

  el("btnTrue").disabled = true;
  el("btnFalse").disabled = true;

  // Answer feedback box
  const answerBox = el("answerBox");
  answerBox.classList.remove("hidden");
  if (isCorrect) {
    answerBox.classList.add("bg-emerald-100", "text-emerald-900", "border-2", "border-emerald-400");

  }

  el("reveal").disabled = true;
  el("next").disabled = false;
}

function tick() {
  const seconds = Math.max(0, Math.ceil((deadline - Date.now()) / 1000));
  const timerEl = el("timer");
  timerEl.querySelector("span").textContent = seconds;

  if (seconds <= 10 && seconds > 0) {
    timerEl.classList.add("text-rose-600", "animate-ping");
    SoundEngine.playTick();
  } else {
    timerEl.classList.remove("text-rose-600", "animate-ping");
  }

  if (seconds === 0) {
    finish(false);
  }
}

  el("score").textContent = `⭐ 0`;
  el("comboBadge").classList.add("hidden");
  el("start").classList.add("hidden");
  el("restart").classList.add("hidden");
  el("reveal").classList.remove("hidden");
  el("next").classList.remove("hidden");

  showQuestion();
  tick();
  interval = setInterval(tick, 200);
}

function finish(completed) {
  clearInterval(interval);
  running = false;
  el("progressBar").style.width = `100%`;

  if (completed) {
    updateMascot("victory");
    SoundEngine.playVictory();
    Confetti.burst();
  } else {
    updateMascot("wrong");
    SoundEngine.playWrong();
  }

  el("progress").textContent = "Rumusan Pencapaian";
  el("icon").textContent = completed ? "🏆" : "⏰";
  el("statement").textContent = completed
    ? `Syabas & Tahniah! Anda Lulus Sebagai Doktor Muda Cilik dengan Markah ⭐ ${score}!`
    : `Masa telah tamat! Mari semak jawapan bersama-sama.`;

  el("interactiveOptions").hidden = true;

  // Generate cheerful certificate & summary card
  const answerBox = el("answerBox");
  answerBox.classList.remove("hidden");
  answerBox.className = "w-full mt-2 p-5 rounded-3xl bg-gradient-to-br from-yellow-50 to-teal-50 border-4 border-yellow-400 text-left text-slate-800 shadow-xl animate-pop-zoom text-sm md:text-base";

  let summaryHTML = `
    <div class="text-center mb-4">
      <span class="text-3xl">🏅</span>
      <h3 class="font-black text-teal-800 text-2xl">SIJIL DOKTOR MUDA CILIK</h3>
      <p class="text-xs text-slate-500 font-bold">Dianugerahkan kepada wira cilik yang bersemangat!</p>
    </div>
    <div class="space-y-2 max-h-60 overflow-y-auto pr-1">
  `;

  questions.forEach((q, i) => {
    const userRes = userAnswers[i];
    const wasCorrect = userRes ? userRes.correct : false;
    const icon = wasCorrect ? "✅" : "❌";
    
    summaryHTML += `
      <div class="p-3 rounded-2xl ${wasCorrect ? 'bg-emerald-100/80 border border-emerald-300' : 'bg-rose-100/80 border border-rose-300'} flex items-start gap-3 shadow-sm">
        <span class="text-2xl">${icon}</span>
        <div>
          <p class="font-black text-slate-800">${i + 1}. ${q.text}</p>
          <p class="text-xs md:text-sm text-slate-600 font-medium mt-0.5">${q.explanation}</p>
        </div>
      </div>
    `;
  });

  summaryHTML += `</div>`;
  answerBox.innerHTML = summaryHTML;

  el("reveal").classList.add("hidden");
  el("next").classList.add("hidden");
  el("restart").classList.remove("hidden");
}

el("start").onclick = startGame;
el("restart").onclick = startGame;

el("reveal").onclick = () => {
  if (!running || answeredCurrent) return;
  const q = questions[index];
  submitAnswer(q.correct);
};

el("next").onclick = () => {
  if (!running || el("next").disabled) return;
  SoundEngine.playClick();
  if (index === questions.length - 1) {
    finish(true);
  } else {
    index++;
    showQuestion();
  }
};

el("fullscreen").onclick = async () => {
  try {
    if (!document.fullscreenElement) {
      await document.documentElement.requestFullscreen();
    } else {
      await document.exitFullscreen();
    }
  } catch (err) {
    console.log("Fullscreen unavailable");
  }
};
